Web Development Security Specifications for Academic Portfolio

Authentication Session Management Owner Permissions Data Protection Implementation

Executive Summary

This technical specification outlines the security architecture for an Academic Portfolio platform. The system implements role-based access control (RBAC) with three permission levels: Owner (Admin), Verified Student, and Public Visitor.

Core Security Objectives

  • Ensure only the verified site owner can access administrative settings
  • Protect draft projects from unauthorized viewing
  • Safeguard personal data with encryption and access controls
  • Implement secure session management with anti-hijacking measures
  • Provide audit logging for all administrative actions

1.3 Owner Verification Code

// Owner Verification Middleware
const verifyOwner = async (req, res, next) => {
    const token = req.headers.authorization?.split(' ')[1];
    
    if (!token) {
        return res.status(401).json({ 
            error: 'Access denied. No token provided.' 
        });
    }

    try {
        const decoded = jwt.verify(token, process.env.JWT_SECRET);
        const user = await User.findById(decoded.userId);

        if (!user) {
            return res.status(401).json({ 
                error: 'Invalid token. User not found.' 
            });
        }

        // CRITICAL: Verify this is the owner
        if (user.role !== 'owner' || user.id !== process.env.OWNER_ID) {
            return res.status(403).json({ 
                error: 'Forbidden. Owner access only.' 
            });
        }

        // Check if account is locked
        if (user.isLocked) {
            return res.status(423).json({ 
                error: 'Account temporarily locked.' 
            });
        }

        req.user = user;
        next();
    } catch (error) {
        return res.status(401).json({ 
            error: 'Invalid or expired token.' 
        });
    }
};

3.2 Owner Verification Code

// Middleware: Owner-Only Access Guard
const ownerOnly = async (req, res, next) => {
    try {
        const user = req.user;

        // Layer 1: Check role
        if (user.role !== 'owner') {
            await logSecurityEvent({
                type: 'UNAUTHORIZED_ACCESS_ATTEMPT',
                userId: user.id,
                endpoint: req.originalUrl,
                ip: req.ip
            });
            return res.status(403).json({
                error: 'Access restricted to site owner'
            });
        }

        // Layer 2: Verify owner ID matches environment
        if (user._id.toString() !== process.env.OWNER_OBJECT_ID) {
            return res.status(403).json({
                error: 'Owner verification failed'
            });
        }

        // Layer 3: Check 2FA status for sensitive operations
        if (req.requires2FA && !user.twoFactorVerified) {
            return res.status(403).json({
                error: 'Two-factor authentication required',
                requires2FA: true
            });
        }

        // Layer 4: Verify request origin
        const allowedOrigins = [
            process.env.PRODUCTION_URL,
            'http://localhost:3000'
        ];
        if (!allowedOrigins.includes(req.headers.origin)) {
            return res.status(403).json({
                error: 'Request origin not allowed'
            });
        }

        next();
    } catch (error) {
        next(error);
    }
};

// Usage in routes

router.put('/admin/update', verifyToken, ownerOnly, updateSettings);

3.3 Draft Project Protection

// Draft Project Access Control
const projectAccessControl = {
    // Middleware to protect draft projects
    checkProjectAccess: async (req, res, next) => {
        const project = await Project.findById(req.params.id);
        
        if (!project) {
            return res.status(404).json({ error: 'Project not found' });
        }

        // Public projects - anyone can view
        if (project.status === 'published' && project.visibility === 'public') {
            return next();
        }

        // Draft/Unlisted - only owner can access
        if (project.status === 'draft' || project.visibility === 'private') {
            if (!req.user || req.user.role !== 'owner') {
                return res.status(404).json({ error: 'Project not found' });
                // Note: Return 404 instead of 403 to hide existence
            }
        }

        // University-only - verified students and owner
        if (project.visibility === 'university') {
            if (!req.user || !['owner', 'verified_student'].includes(req.user.role)) {
                return res.status(404).json({ error: 'Project not found' });
            }
        }

        next();
    }
};

5. Implementation Reference

5.1 Backend Security Middleware Stack

// Complete Security Middleware Stack
const securityMiddleware = [
    // 1. Rate Limiting
    rateLimit({
        windowMs: 15 * 60 * 1000, // 15 minutes
        max: 100, // limit each IP to 100 requests per windowMs
        message: 'Too many requests, please try again later'
    }),

    // 2. CORS Configuration
    cors({
        origin: process.env.ALLOWED_ORIGINS.split(','),
        credentials: true,
        methods: ['GET', 'POST', 'PUT', 'DELETE'],
        allowedHeaders: ['Content-Type', 'Authorization']
    }),

    // 3. Helmet Security Headers
    helmet({
        contentSecurityPolicy: {
            directives: {
                defaultSrc: ["'self'"],
                scriptSrc: ["'self'", "'unsafe-inline'"],
                styleSrc: ["'self'", "'unsafe-inline'", "https://fonts.googleapis.com"],
                imgSrc: ["'self'", "data:", "https:"],
            }
        },
        hsts: {
            maxAge: 31536000,
            includeSubDomains: true,
            preload: true
        }
    }),

    // 4. Request Validation
    expressValidator({
        customValidators: {
            isOwner: (value) => value === process.env.OWNER_ID,
            isValidToken: (token) => jwt.verify(token, process.env.JWT_SECRET)
        }
    }),

    // 5. Security Logging
    morgan('combined', {
        stream: logger.stream,
        skip: (req, res) => res.statusCode < 400
    })
];

5.2 Frontend Route Protection

// React Route Guards
const ProtectedRoute = ({ children, requiredRole }) => {
    const { user, isAuthenticated } = useAuth();
    const location = useLocation();

    if (!isAuthenticated) {
        return <Navigate to="/login" state={{ from: location }} />;
    }

    if (requiredRole && user.role !== requiredRole) {
        return <Navigate to="/unauthorized" />;
    }

    return children;
};

// Usage
<Route path="/admin/*" element={
    <ProtectedRoute requiredRole="owner">
        <AdminDashboard />
    </ProtectedRoute>
} />

<Route path="/drafts/*" element={
    <ProtectedRoute requiredRole="owner">
        <DraftProjects />
    </ProtectedRoute>
} />
